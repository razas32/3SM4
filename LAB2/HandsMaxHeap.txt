import java.util.Random;

public class HandsMaxHeap {
    private Hands[] myHeap;  // array to hold the heap - using array-based implementation like we learned in class
    private int size;      // how many valid hands I actually have stored in the heap
    private int capacity;  // the max number of hands this heap can hold before I need to expand

    /* Constructor 1: Start with an empty heap that has room for bufSize hands
     * Time: O(1) since I'm just setting up the array and a dummy node
     * Space: O(bufSize) for the array */
    public HandsMaxHeap(int bufSize)
    {
        // starting with an empty heap
        this.capacity = bufSize;
        this.size = 0;

        // need +1 slots because index 0 is a dummy (makes parent/child math cleaner: parent = i/2, left = 2*i, right = 2*i+1)
        this.myHeap = new Hands[capacity + 1];

        // stick a dummy invalid hand at index 0 - never gonna use it but it simplifies the indexing
        this.myHeap[0] = new Hands();
    }

    // Constructor 2: constructs a heap out of the array someHands 
    // the first element in the array is treated as a dummy, the remaining elements are organized as a heap using 
    // the private method bulidMaxHeap, which you have to implement
    
    public HandsMaxHeap(Hands[] someHands)
    {
        myHeap = someHands;
        capacity = someHands.length - 1;
        size = capacity;
        buildMaxHeap();  
    }

    // [Problem 0] Implement buildMaxHeap

    /* Build a heap from an unsorted array - way faster than inserting one by one!
     * Start from the last internal node and work backwards, calling downHeapify on each
     * Time: O(n) which is actually amazing compared to O(n log n) for inserting each element separately
     * Space: O(1) since I'm modifying in place */
    private void buildMaxHeap()
    {
        // last internal node is at size/2 - everything after that is just leaves
        // leaves are already valid heaps, so start from the last non-leaf and work backwards

        for(int i = size / 2; i >= 1; i--)
        {
            downHeapify(i);  // sink this node down to its proper spot
        }

        // done! whole array is now a proper max heap
    }
  
    // [Problem 1] Implement Max Heap for 5-Card Hands

    // [Problem 1-1] Implement Private Utility Methods

    /* Find the parent of a node - standard array heap formula
     * Time: O(1) just integer division
     * Space: O(1) */
    private int parent(int index)
    {
        // parent is always at index/2 in an array heap
        // e.g. node 4 and node 5 both have parent at index 2
        return index / 2;
    }

    /* Find left child in the array
     * Time: O(1)
     * Space: O(1) */
    private int leftChild(int index)
    {
        // left child is at 2*index, so node 3 has left child at 6
        return 2 * index;
    }

    /* Find right child
     * Time/Space: both O(1) */
    private int rightChild(int index)
    {
        // right child at 2*index + 1, e.g. node 3 has right child at 7
        return 2 * index + 1;
    }

    // [Problem 1-2] Implement the Downward Heap Reorganization Private Method from the provided index
    // this is the percolateDown discussed in class

    /* Push a hand downward to its correct spot
     * Use this after removing max - take the last element, stick it at root, then sink it
     * Time: O(log n) could go all the way down to the leaves
     * Space: O(1) */
    private void downHeapify(int index)
    {
        Hands temp = myHeap[index];  // save what I'm sinking

        // keep going while there's at least a left child
        while(leftChild(index) <= size)
        {
            int left = leftChild(index);
            int right = rightChild(index);
            int largerChild = left;  // start by assuming left is bigger

            // does right child exist and is it actually bigger?
            if(right <= size && myHeap[right].isMyHandLarger(myHeap[left]))
            {
                largerChild = right;
            }

            // am I bigger than or equal to the larger child? done!
            if(temp.isMyHandLarger(myHeap[largerChild]) || temp.isMyHandEqual(myHeap[largerChild]))
            {
                break;
            }

            // child is bigger, pull it up
            myHeap[index] = myHeap[largerChild];

            // keep sinking
            index = largerChild;
        }

        myHeap[index] = temp;  // final resting place
    } 

    // [Problem 1-3] Implement Upward Heap Reorganization Private Method from the provided index
    // this is the percolateUp discussed in class

    /* Bubble a hand upward until it finds where it belongs
     * Called after inserting at the end to restore max heap property
     * Time: O(log n) because worst case I go all the way up to the root
     * Space: O(1) just swapping elements around */
    private void heapifyUp(int index)
    {
        // save the hand I'm placing so I can compare it as I move up
        Hands temp = myHeap[index];

        // keep bubbling up until I either hit the root or find a parent that's actually bigger
        while(index > 1 && myHeap[parent(index)].isMyHandSmaller(temp))
        {
            // parent is weaker, so pull it down
            myHeap[index] = myHeap[parent(index)];

            // move up to parent's old spot
            index = parent(index);
        }

        // okay found where this hand belongs
        myHeap[index] = temp;
    }


    // [Problem 1-4] Complete the Max Heap ADT Public Methods

    /* Insert a new hand into the heap
     * Time: O(log n) typically, O(n) if we need to resize
     * Space: O(n) when resizing (gotta copy everything), otherwise O(1) */
    public void insert(Hands thisHand)
    {
        // check if we're out of room
        if(size >= capacity)
        {
            // double the capacity to amortize resize cost
            capacity = capacity * 2;
            Hands[] newHeap = new Hands[capacity + 1];

            // copy over existing elements (including dummy at index 0)
            for(int i = 0; i <= size; i++)
            {
                newHeap[i] = myHeap[i];
            }

            myHeap = newHeap;  // switch to bigger array
        }

        // got space - stick the new hand at the end
        size++;
        myHeap[size] = thisHand;

        // bubble it up to restore heap property
        heapifyUp(size);
    }

    /* Remove and return the strongest hand (always at the root)
     * The aggressive AI uses this to always pick the best available hand
     * Time: O(log n) because of the downHeapify
     * Space: O(1) */
    public Hands removeMax() throws RuntimeException
    {
        // can't remove from empty heap
        if(size == 0)
        {
            throw new RuntimeException("Heap is empty! Can't remove from nothing.");
        }

        Hands maxHand = myHeap[1];  // grab the root (always the max)

        // move last element to root
        myHeap[1] = myHeap[size];
        size--;

        // sink it down to restore heap property (if heap isn't empty now)
        if(size > 0)
        {
            downHeapify(1);
        }

        return maxHand;
    }

    /* Get the number of hands currently in heap
     * Time/Space: O(1) */
    public int getSize()
    {
        return size;
    }

    /* Check if heap is empty
     * Time/Space: O(1) */
    public boolean isEmpty()
    {
        return size == 0;
    }

    /* Print the heap contents in level order - useful for debugging
     * Needed for the demo to show what hands the AI has
     * Time: O(n) visit each element once
     * Space: O(1) */
    public void printHeap()
    {
        System.out.println("\n========== HEAP CONTENTS ==========");
        System.out.println("Heap size: " + size);

        // go through heap from 1 to size (level order comes free with array implementation)
        for(int i = 1; i <= size; i++)
        {
            System.out.print("  [" + i + "] ");

            if(myHeap[i].isAValidHand())
            {
                myHeap[i].printMyHand();
                System.out.println();
            }
            else
            {
                System.out.println("--INV--");
            }
        }

        System.out.println("===================================\n");
    }

    /* Sort array IN PLACE in DESCENDING order with heap sort
     * IN PLACE = no extra arrays, just rearrange what's there
     * Time: O(n log n) build heap in O(n), then n extractions at O(log n) each
     * Space: O(1) */
    public static void heapSort(Hands myHands[])
    {
        int n = myHands.length;

        // build a max heap from the array (using 0-indexing not 1-indexing like my class)
        // start from last internal node and work backwards
        for(int i = n / 2 - 1; i >= 0; i--)
        {
            heapifyDown(myHands, n, i);
        }

        // now repeatedly swap root with last element and re-heapify
        // builds sorted array from right to left
        for(int i = n - 1; i > 0; i--)
        {
            // swap max to the end
            Hands temp = myHands[0];
            myHands[0] = myHands[i];
            myHands[i] = temp;

            // max is now in final position, shrink heap and fix
            heapifyDown(myHands, i, 0);
        }

        // array sorted in descending order!
    }

    /* Helper for heapSort - percolate down in 0-indexed array
     * Similar to my downHeapify but for arrays starting at 0
     * Uses MIN heap for descending sort (smallest bubbles up and gets moved to end)
     * Time: O(log n), Space: O(1) */
    private static void heapifyDown(Hands[] arr, int heapSize, int index)
    {
        Hands temp = arr[index];

        while(2 * index + 1 < heapSize)  // while there's a left child
        {
            int left = 2 * index + 1;
            int right = 2 * index + 2;
            int smallerChild = left;

            // is right child smaller? (min heap logic)
            if(right < heapSize && arr[right].isMyHandSmaller(arr[left]))
            {
                smallerChild = right;
            }

            // am I already smaller than smallest child? done
            if(temp.isMyHandSmaller(arr[smallerChild]) || temp.isMyHandEqual(arr[smallerChild]))
            {
                break;
            }

            // pull smaller child up
            arr[index] = arr[smallerChild];
            index = smallerChild;
        }

        arr[index] = temp;
    }

    private static boolean assertEquals(Hands a, Hands b)
    {
        if(!a.isMyHandEqual(b))
        {
            System.out.println("\tAssert Failed!");
            System.out.printf("\tExpected: ");
            a.printMyHand();
            System.out.printf(", Actual: ");
            b.printMyHand();
            System.out.printf("\n");
            return false;
        }

        return true;
    }

    private static boolean assertEquals(int a, int b)
    {
        if(a != b)
        {
            System.out.println("\tAssert Failed!");
            System.out.printf("\tExpected: %d, Actual: %d\n\n", a, b);
            return false;
        }

        return true;
    }

    private static boolean assertEquals(boolean a, boolean b)
    {
        if(a != b)
        {
            System.out.println("\tAssert Failed!");
            System.out.printf("\tExpected: %b, Actual: %b\n\n", a, b);
            return false;
        }

        return true;
    }
